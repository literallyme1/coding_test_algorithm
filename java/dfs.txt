boolean visited = new boolean[N][M];
int[] dr = {1, -1, 0, 0};
int[] dc = {0, 0, -1, 1};

public dfs(r, c){

    visited[r][c] = true;

    for(int i = 0; i < 4; i ++){
        int nr = r + dr[i];
        int nc = c + dc[i];

        if(0 > nr || nr >= N|| 0 > nc || nc >= M){
            continue;
        }
        if(map[nr][nc] == 0){
            continue;
        }

        if(visited[nr][nc]) { continue; }
        dfs(nr, nc);
        
    }
}
