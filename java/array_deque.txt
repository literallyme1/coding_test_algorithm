Deque<Integer> deque = new ArrayDeque<>();

deque.offerFirst();
deque.offerLast();
deque.peekFirst();
deque.peekLast();
deque.pollFirst();
deque.pollLast();

deque.size();
deque.contains();
deque.isEmpty();

for(String value : str.toCharArray()){
    if(value == '('){
        deque.offerLast(value);
    }else{
        if(deque.isEmpty()){
            return false;
        }
        deque.pollLast();
    }
}
