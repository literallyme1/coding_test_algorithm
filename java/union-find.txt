int[] parent = new int[N+1];
for(1){
    parent[i] = i;
}



int find(int a){
    if(parent[a] == a){
        return a;
    }
    return parent[a] = find(parent[a]);

}

boolean union(a, b){
    int rootA = find(a);
    int rootB = find(b);
    if(rootA == rootB) {
        return false;
    }

    parent[rootB] = rootA;
    return true;
}
